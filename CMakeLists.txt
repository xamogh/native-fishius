cmake_minimum_required(VERSION 3.24)
project(FishXAquarium VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(AQ_BUILD_APP "Build the SDL3 native game" ON)
option(AQ_BUILD_TESTS "Build SDL-independent domain tests" ON)
option(AQ_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
option(AQ_FETCH_DEPENDENCIES "Download pinned dependencies if missing" ON)
include(FetchContent)
find_package(nlohmann_json 3.11 CONFIG QUIET)
if(NOT TARGET nlohmann_json::nlohmann_json)
 if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/nlohmann/json.hpp")
  add_library(aq_json INTERFACE)
  target_include_directories(aq_json INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/third_party")
  add_library(nlohmann_json::nlohmann_json ALIAS aq_json)
 elseif(AQ_FETCH_DEPENDENCIES)
  FetchContent_Declare(json URL https://codeload.github.com/nlohmann/json/tar.gz/refs/tags/v3.11.3 DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(json)
 else()
  message(FATAL_ERROR "Install nlohmann_json 3.11.3 or enable dependency fetching")
 endif()
endif()
add_library(aquarium_domain src/domain.cpp)
target_include_directories(aquarium_domain PUBLIC include)
target_link_libraries(aquarium_domain PUBLIC nlohmann_json::nlohmann_json)
add_library(aquarium_storage src/storage.cpp)
target_link_libraries(aquarium_storage PUBLIC aquarium_domain)
function(aq_options target)
 if(MSVC)
  target_compile_options(${target} PRIVATE /W4 /permissive-)
 else()
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wshadow)
  if(AQ_SANITIZERS)
   target_compile_options(${target} PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
   target_link_options(${target} PUBLIC -fsanitize=address,undefined)
  endif()
 endif()
endfunction()
aq_options(aquarium_domain)
aq_options(aquarium_storage)
if(AQ_BUILD_APP)
 find_package(SDL3 3.2 CONFIG QUIET)
 if(NOT TARGET SDL3::SDL3)
  if(NOT AQ_FETCH_DEPENDENCIES)
   message(FATAL_ERROR "SDL3 3.2.20 is required")
  endif()
  set(SDL_TESTS OFF CACHE BOOL "" FORCE)
  set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
  set(SDL_EXAMPLES OFF CACHE BOOL "" FORCE)
  if(ANDROID)
   set(SDL_SHARED ON CACHE BOOL "" FORCE)
   set(SDL_STATIC OFF CACHE BOOL "" FORCE)
  else()
   set(SDL_SHARED OFF CACHE BOOL "" FORCE)
   set(SDL_STATIC ON CACHE BOOL "" FORCE)
  endif()
  FetchContent_Declare(SDL3 URL https://codeload.github.com/libsdl-org/SDL/tar.gz/refs/tags/release-3.2.20 DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(SDL3)
 endif()
 find_package(SDL3_image 3.2 CONFIG QUIET)
 if(NOT TARGET SDL3_image::SDL3_image)
  set(SDLIMAGE_SAMPLES OFF CACHE BOOL "" FORCE)
  set(SDLIMAGE_TESTS OFF CACHE BOOL "" FORCE)
  set(SDLIMAGE_AVIF OFF CACHE BOOL "" FORCE)
  set(SDLIMAGE_JXL OFF CACHE BOOL "" FORCE)
  set(SDLIMAGE_WEBP OFF CACHE BOOL "" FORCE)
  set(SDLIMAGE_TIF OFF CACHE BOOL "" FORCE)
  set(SDLIMAGE_VENDORED OFF CACHE BOOL "" FORCE)
  if(IOS)
   set(SDLIMAGE_BACKEND_IMAGEIO OFF CACHE BOOL "Use the portable image loader on iOS" FORCE)
  endif()
  FetchContent_Declare(SDL3_image URL https://codeload.github.com/libsdl-org/SDL_image/tar.gz/refs/tags/release-3.2.4 DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(SDL3_image)
 endif()
 find_package(SDL3_ttf 3.2 CONFIG QUIET)
 if(NOT TARGET SDL3_ttf::SDL3_ttf)
  set(SDLTTF_SAMPLES OFF CACHE BOOL "" FORCE)
  set(SDLTTF_TESTS OFF CACHE BOOL "" FORCE)
  set(SDLTTF_VENDORED OFF CACHE BOOL "" FORCE)
  set(SDLTTF_HARFBUZZ OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(SDL3_ttf URL https://codeload.github.com/libsdl-org/SDL_ttf/tar.gz/refs/tags/release-3.2.2 DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(SDL3_ttf)
 endif()
 add_library(aquarium_view src/canvas.cpp src/view.cpp src/skin.cpp)
 target_link_libraries(aquarium_view PUBLIC aquarium_storage SDL3::SDL3 SDL3_image::SDL3_image SDL3_ttf::SDL3_ttf)
 aq_options(aquarium_view)
 if(ANDROID)
  add_library(main SHARED src/main.cpp)
  target_link_libraries(main PRIVATE aquarium_view)
  set(AQ_EXECUTABLE main)
 elseif(IOS)
  add_executable(aquarium MACOSX_BUNDLE src/main.cpp)
  target_link_libraries(aquarium PRIVATE aquarium_view)
  set_target_properties(aquarium PROPERTIES MACOSX_BUNDLE_GUI_IDENTIFIER "com.pixmot.aquarium" MACOSX_BUNDLE_BUNDLE_NAME "FishX" MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/platform/ios/Info.plist" XCODE_ATTRIBUTE_TARGETED_DEVICE_FAMILY "1,2")
  # Copy the whole asset tree after every link. A configure-time glob would
  # miss assets regenerated after the build tree was configured, and would
  # silently drop file types absent from its pattern list.
  add_custom_command(TARGET aquarium POST_BUILD
   COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/assets" "$<TARGET_BUNDLE_DIR:aquarium>/assets")
  set(AQ_EXECUTABLE aquarium)
 else()
  add_executable(aquarium src/main.cpp)
  target_link_libraries(aquarium PRIVATE aquarium_view)
  add_custom_command(TARGET aquarium POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/assets" "$<TARGET_FILE_DIR:aquarium>/assets")
  install(TARGETS aquarium RUNTIME DESTINATION .)
  install(DIRECTORY assets DESTINATION . PATTERN "workbook.json" EXCLUDE)
  set(AQ_EXECUTABLE aquarium)
 endif()
 # Asset edits must refresh the installed app even when no C++ file changed.
 file(GLOB_RECURSE AQ_ASSET_FILES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/assets/*")
 set_property(TARGET ${AQ_EXECUTABLE} APPEND PROPERTY LINK_DEPENDS "${AQ_ASSET_FILES}")
 aq_options(${AQ_EXECUTABLE})
endif()

  add_test(NAME mobile_ui_controls COMMAND aquarium_view_tests "${CMAKE_CURRENT_SOURCE_DIR}/assets")
  set_tests_properties(mobile_ui_controls PROPERTIES LABELS "native;ui")
 endif()
endif()
